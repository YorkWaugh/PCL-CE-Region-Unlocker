cmake_minimum_required(VERSION 3.24)
project(PclCeRegionUnlocker VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC on Windows (amd64 or ARM64).")
endif()
if(NOT CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|ARM64)$")
  message(FATAL_ERROR "Only amd64 (-A x64) and ARM64 (-A ARM64) are supported.")
endif()
configure_file(src/version.rc.in ${CMAKE_CURRENT_BINARY_DIR}/version.rc @ONLY)
add_library(winmm SHARED src/proxy.cpp src/profiler.cpp src/winmm.cpp src/winmm.def
  ${CMAKE_CURRENT_BINARY_DIR}/version.rc)
foreach(target winmm)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_include_directories(${target} PRIVATE vendor/dotnet)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
  target_compile_options(${target} PRIVATE /W4 /wd4100 /permissive- /EHsc /utf-8)
  set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_link_options(${target} PRIVATE /DYNAMICBASE /NXCOMPAT)
  target_link_libraries(${target} PRIVATE ole32)
  set_target_properties(${target} PROPERTIES PREFIX "")
endforeach()
